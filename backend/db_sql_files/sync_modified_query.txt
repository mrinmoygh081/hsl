
CREATE TABLE IF NOT EXISTS public.cron_log
(
    cron_name character varying(200) COLLATE pg_catalog."default" NOT NULL,
    execute_datetime timestamp with time zone NOT NULL,
    status character varying(100) COLLATE pg_catalog."default" NOT NULL,
    created_by character varying(12) COLLATE pg_catalog."default",
    remarks text COLLATE pg_catalog."default",
    error text COLLATE pg_catalog."default",
    sync boolean,
    sync_id uuid NOT NULL DEFAULT gen_random_uuid(),
    sync_updated_at timestamp with time zone NOT NULL DEFAULT now(),
    CONSTRAINT cron_log_sync_id_key UNIQUE (sync_id)
)

----------------------------------------------------


-- Ensure pgcrypto for gen_random_uuid()
CREATE EXTENSION IF NOT EXISTS pgcrypto;

DO
$$
DECLARE
    r RECORD;
BEGIN
    -- Loop over all tables in the public schema
    FOR r IN
        SELECT table_name
        FROM information_schema.tables
        WHERE table_schema = 'public'
          AND table_type = 'BASE TABLE'
    LOOP
        -- Add column sync_id if it does not exist
        IF NOT EXISTS (
            SELECT 1
            FROM information_schema.columns
            WHERE table_schema = 'public'
              AND table_name   = r.table_name
              AND column_name  = 'sync_id'
        ) THEN
            EXECUTE format('ALTER TABLE public.%I ADD COLUMN sync_id UUID DEFAULT gen_random_uuid() UNIQUE NOT NULL;', r.table_name);
        END IF;

        -- Add column sync if it does not exist
        IF NOT EXISTS (
            SELECT 1
            FROM information_schema.columns
            WHERE table_schema = 'public'
              AND table_name   = r.table_name
              AND column_name  = 'sync'
        ) THEN
            EXECUTE format('ALTER TABLE public.%I ADD COLUMN sync BOOLEAN DEFAULT FALSE NOT NULL;', r.table_name);
        END IF;

        -- Add column sync_updated_at if it does not exist
        IF NOT EXISTS (
            SELECT 1
            FROM information_schema.columns
            WHERE table_schema = 'public'
              AND table_name   = r.table_name
              AND column_name  = 'sync_updated_at'
        ) THEN
            EXECUTE format('ALTER TABLE public.%I ADD COLUMN sync_updated_at TIMESTAMPTZ DEFAULT now() NOT NULL;', r.table_name);
        END IF;
    END LOOP;
END
$$;
